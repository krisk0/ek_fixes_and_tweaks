# ek_fixes_and_tweaks modifications:
#  new-born vampire heals of most diseases
#  code de-duplicated
#  minor optimizations
give_vampire_secret_or_trait_effect = {
	if = {
		limit = { #Can become vampire
			NOR = {
				# ek_fixes_and_tweaks: definite condition for being vampire character is
				#  vampire character flag, EK code checked two conditions instead
				has_character_flag = vampire
				has_character_flag = race_daedra # daedra cant become vampires
			}
		}
		if = {
			# ek_fixes_and_tweaks: scope definition moved here
			save_temporary_scope_as = vampire_character
			# ek_fixes_and_tweaks: whitespace change
			limit = {
				trait_is_shunned_or_criminal_in_my_or_lieges_faith_trigger = {
					TRAIT = trait:vampire_character
					GENDER_CHARACTER = scope:vampire_character
				}
			}
			add_secret = { type = secret_vampire }
			add_trait = reclusive_vampire
			add_character_modifier = secret_vampire_modifier
		}
		else = {
			add_trait = vampire_character
		}
		# ek_fixes_and_tweaks: common code moved here
		retain_visual_age_effect = yes #always do this before adding the immortality trait
		add_character_flag = vampire
		substitute_lifespan_by_immortality_effect = yes #always do this after adding the immortality trait
		custom_tooltip = {
			text = ek_clear_aging_modifier_tt
			clear_aging_modifier_effect = yes
		}
		# ek_fixes_and_tweaks: heal most diseases, according to Skyrim lore
		magical_cure_disease = yes
	}
}
